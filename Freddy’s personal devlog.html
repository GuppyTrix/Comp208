  unsigned long now = millis();
  float deltatime = (now - lasttime) / 1000.0;
  if (deltatime <= 0)
  {
    deltatime = 0.0001;
  }

  long lticks = myEncoderL.read();
  long rticks = myEncoderR.read();

  long ldeltaticks = lticks - lprevticks;
  long rdeltaticks = rticks - rprevticks;

  float wheelradius = 0.00;
  float angular_velocityw1 = (ldeltaticks / ticksPerDegree) * (PI / 180.0) / dt;
  float angular_velocityw2 = (rdeltaticks / ticksPerDegree) * (PI / 180.0) / dt;

#include <Pixy2.h>
#include <Pixy2CCC.h>

Pixy2 pixy;

// Motor pins
const int motor3a = 38;
const int motor3b = 40;
const int motor1a = 41;
const int motor1b = 39;
const int motor2a = 45;
const int motor2b = 43;
const int motor4a = 42;
const int motor4b = 44;

// Tracking variables
float deadzone = 0.15;
int signature, x, y, width, height;
float cx;
//state machine
enum State { SEARCH, TRACK_CAN, AVOID, CENTERED, TURN };
State state = SEARCH;

void setup()
{
  Serial.begin(115200);
  pixy.init();
  pixy.changeProg("color_connected_components");

  pinMode(motor1a, OUTPUT);
  pinMode(motor1b, OUTPUT);
  pinMode(motor2a, OUTPUT);
  pinMode(motor2b, OUTPUT);
  pinMode(motor3a, OUTPUT);
  pinMode(motor3b, OUTPUT);
  pinMode(motor4a, OUTPUT);
  pinMode(motor4b, OUTPUT);
}

// Movement functions
void forward() {
  digitalWrite(motor1a, HIGH); digitalWrite(motor1b, LOW);
  digitalWrite(motor2a, HIGH); digitalWrite(motor2b, LOW);
  digitalWrite(motor3a, HIGH); digitalWrite(motor3b, LOW);
  digitalWrite(motor4a, HIGH); digitalWrite(motor4b, LOW);
}

void backward() {
  digitalWrite(motor1a, LOW); digitalWrite(motor1b, HIGH);
  digitalWrite(motor2a, LOW); digitalWrite(motor2b, HIGH);
  digitalWrite(motor3a, LOW); digitalWrite(motor3b, HIGH);
  digitalWrite(motor4a, LOW); digitalWrite(motor4b, HIGH);
}

void stopMotors() {
  digitalWrite(motor1a, LOW); digitalWrite(motor1b, LOW);
  digitalWrite(motor2a, LOW); digitalWrite(motor2b, LOW);
  digitalWrite(motor3a, LOW); digitalWrite(motor3b, LOW);
  digitalWrite(motor4a, LOW); digitalWrite(motor4b, LOW);
}

// Mapping
float mapfloat(long x,long in_min,long in_max,long out_min,long out_max)
{
  return float(x - in_min) * (out_max - out_min) / float(in_max - in_min) + out_min;
}

// Read Pixy, return signature or 0 if none for debugging
int checkPixy()
{
  uint16_t blocks = pixy.ccc.getBlocks();

  if(blocks == 0) return 0;

  signature = pixy.ccc.blocks[0].m_signature;
  x = pixy.ccc.blocks[0].m_x;
  width = pixy.ccc.blocks[0].m_width;

  cx = x + (width / 2);
  cx = mapfloat(cx, 0, 316, -1, 1);

  return signature;
}

// AVOID BEHAVIOUR
void avoidObstacle()
{
  Serial.println("AVOIDING");

  // Stop then back up then turn right
  stopMotors();
  delay(150);

  backward();
  delay(250);

  // turn away from obstacle
  digitalWrite(motor1a, HIGH); digitalWrite(motor1b, LOW);
  digitalWrite(motor2a, LOW);  digitalWrite(motor2b, HIGH);
  digitalWrite(motor3a, HIGH); digitalWrite(motor3b, LOW);
  digitalWrite(motor4a, LOW);  digitalWrite(motor4b, HIGH);

  delay(350);

  stopMotors();
}

void loop()
{
  int sig = checkPixy();

  // No detections == SEARCH
  if(sig == 0)
  {
    state = SEARCH;
  }
  // Signature 2 = obstacle == AVOID
  else if(sig == 2)
  {
    state = AVOID;
  }
  // Signature 1 = can to chase
  else if(sig == 1)
  {
    state = TRACK_CAN;
  }

  // STATES AND THEIR ACTIONS
  switch(state)
  {
    case SEARCH:
      stopMotors();
      break;

    case AVOID:
      avoidObstacle();
      state = SEARCH;
      break;

    case TRACK_CAN:
      // if in the centre go forward
      if(cx > -deadzone && cx < deadzone)
      {
        forward();
      }
      // If off-centre, bias movement
      else if(cx < 0)
      {
        forward();   // left bias
      }
      else if(cx > 0)
      {
        backward();  // right bias
      }
      break;
  }

  delay(40);
}
